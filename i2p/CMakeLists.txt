cmake_minimum_required(VERSION 3.20)
project(mukhomor_i2pd LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)
file(GLOB core "${I2PD_SOURCE}/libi2pd/*.cpp")
file(GLOB client "${I2PD_SOURCE}/libi2pd_client/*.cpp")
file(GLOB languages "${I2PD_SOURCE}/i18n/*.cpp")
if(BOOST_SOURCE)
    file(GLOB options "${BOOST_SOURCE}/libs/program_options/src/*.cpp")
else()
    find_package(Boost REQUIRED COMPONENTS program_options)
endif()
add_executable(mukhomor-i2pd main.cpp ${core} ${client} ${languages} ${options}
    "${I2PD_SOURCE}/daemon/Daemon.cpp" "${I2PD_SOURCE}/daemon/UPnP.cpp"
    "${I2PD_SOURCE}/daemon/HTTPServer.cpp" "${I2PD_SOURCE}/daemon/I2PControl.cpp"
    "${I2PD_SOURCE}/daemon/I2PControlHandlers.cpp")
target_include_directories(mukhomor-i2pd PRIVATE "${BOOST_SOURCE}" "${I2PD_SOURCE}/libi2pd"
    "${I2PD_SOURCE}/libi2pd_client" "${I2PD_SOURCE}/daemon" "${I2PD_SOURCE}/i18n")
target_compile_definitions(mukhomor-i2pd PRIVATE MUKHOMOR_EMBEDDED STD_FILESYSTEM=1
    BOOST_ALL_NO_LIB BOOST_SYSTEM_NO_LIB)
target_compile_options(mukhomor-i2pd PRIVATE -fstack-protector-strong -fno-strict-aliasing
    "-ffile-prefix-map=${I2PD_SOURCE}=/source/i2pd" "-ffile-prefix-map=${BOOST_SOURCE}=/source/boost"
    "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=/source/mukhomor/i2p")
target_link_libraries(mukhomor-i2pd PRIVATE OpenSSL::SSL OpenSSL::Crypto ZLIB::ZLIB Threads::Threads ${CMAKE_DL_LIBS})
if(NOT BOOST_SOURCE)
    target_link_libraries(mukhomor-i2pd PRIVATE Boost::program_options)
endif()
if(WIN32)
    target_compile_definitions(mukhomor-i2pd PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0602 _WIN32_WINNT=0x0602)
    target_link_libraries(mukhomor-i2pd PRIVATE ws2_32 mswsock iphlpapi crypt32)
    target_link_options(mukhomor-i2pd PRIVATE -static -mwindows -Wl,--nxcompat,--dynamicbase,--high-entropy-va)
else()
    target_link_options(mukhomor-i2pd PRIVATE -Wl,-z,relro,-z,now)
    if(NOT ANDROID)
        target_link_options(mukhomor-i2pd PRIVATE -static-libgcc -static-libstdc++)
    endif()
endif()
if(ANDROID)
    target_compile_definitions(mukhomor-i2pd PRIVATE ANDROID)
    target_compile_options(mukhomor-i2pd PRIVATE "-ffile-prefix-map=${CMAKE_ANDROID_NDK}=/source/android-ndk")
    target_link_options(mukhomor-i2pd PRIVATE -Wl,-z,max-page-size=16384)
endif()
